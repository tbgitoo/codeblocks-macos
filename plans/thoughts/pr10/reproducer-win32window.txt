`Win32Window` reproducer

`loadPlugin()` does not reach `Win32Window::create()`. The host in this appendix extends the Appendix A program with a third argument, `GL` or `GL3Plus`, then:

1. loads both plugins in the requested order;
2. selects that render system;
3. calls `Root::initialise(false)`;
4. calls `Root::createRenderWindow()` for one 320x240 windowed window titled `OgreWin32Window module-handle test`;
5. reads the native `HWND` and client size;
6. destroys the window with `Root::destroyRenderTarget()`.

That call chain is:

```text
Root::createRenderWindow
 -> GLRenderSystem::_createRenderWindow
    or GL3PlusRenderSystem::_createRenderWindow
 -> Win32GLSupport::newWindow
 -> Win32Window::create
```

The lookup inside `create()` is `GetModuleHandle("GLSupport.dll")` on this Release build. The host also prints `GetModuleHandle("GLSupport.dll")`. In the tested shared build that module is not loaded, so the lookup is `NULL` for every order and both renderers.

`dllStartPlugin` is still only an export of the render-system DLL. It is not an address inside `OgreWin32Window.cpp`.

static void printModule(const char* name)
{
    HMODULE module = GetModuleHandleA(name);
    std::cout << name << ": " << static_cast<const void*>(module);
    if (!module)
    {
        std::cout << " not loaded" << std::endl;
        return;
    }

    char path[MAX_PATH];
    DWORD length = GetModuleFileNameA(module, path, MAX_PATH);
    if (length == 0 || length >= MAX_PATH)
        std::cout << " loaded, path unavailable" << std::endl;
    else
        std::cout << " loaded " << path << std::endl;

static void printFromAddress(const char* dllName)
{
    HMODULE module = GetModuleHandleA(dllName);
    FARPROC entry = module ? GetProcAddress(module, "dllStartPlugin") : NULL;
    std::cout << dllName << " dllStartPlugin: " << reinterpret_cast<const void*>(entry) << std::endl;
    std::cout << dllName << " FROM_ADDRESS: ";
    HMODULE owner = NULL;
    if (!entry || !GetModuleHandleExA(
            GET_MODULE_HANDLE_EX_FLAG_FROM_ADDRESS | GET_MODULE_HANDLE_EX_FLAG_UNCHANGED_REFCOUNT,
            reinterpret_cast<LPCSTR>(entry),

## `OgreGLWindow` class probe

`Win32Window::create()` registers class `"OgreGLWindow"` with `CS_OWNDC`, `LoadIcon(NULL, IDI_APPLICATION)`, `LoadCursor(NULL, IDC_ARROW)`, a black brush, and window style `WS_VISIBLE | WS_CLIPCHILDREN | WS_OVERLAPPEDWINDOW`. The default procedure is `DefWindowProc`, which lives in `user32.dll`. `hInst` is local and is not stored. `destroy()` calls `DestroyWindow` and does not call `UnregisterClass`.

The external probe repeats one register/create/destroy/unregister cycle for three instance values. It does not use the class name `OgreGLWindow`, so it does not collide with a live OGRE window. It needs a small disposable `probe.dll` only as the unrelated module in case C.

Add these targets beside the host. `probe.cpp`:

```cpp
#include <windows.h>

extern "C" __declspec(dllexport) LRESULT CALLBACK ProbeWndProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp)
{
    return DefWindowProcA(hwnd, msg, wp, lp);
}
```

`win32_report.h` supplies the path, error, and `FROM_ADDRESS` helpers used by `class_ogrewindow.cpp`:

```cpp
#pragma once

#include <windows.h>

#include <fstream>
#include <iostream>
#include <sstream>
#include <string>

inline std::string modulePath(HMODULE module)
{
    if (!module)
        return "(null)";
    char path[MAX_PATH];
    DWORD length = GetModuleFileNameA(module, path, MAX_PATH);
    if (length == 0 || length >= MAX_PATH)
        return "(path unavailable)";
    return path;
}

inline HMODULE moduleFromAddress(const void* address)
{
    HMODULE module = NULL;
    if (!address)
        return NULL;
    if (!GetModuleHandleExA(
            GET_MODULE_HANDLE_EX_FLAG_FROM_ADDRESS | GET_MODULE_HANDLE_EX_FLAG_UNCHANGED_REFCOUNT,
            reinterpret_cast<LPCSTR>(address),
            &module))
        return NULL;
    return module;
}

inline std::string win32Message(DWORD code)
{
    if (code == 0)
        return "0";
    char buffer[512];
    DWORD length = FormatMessageA(
        FORMAT_MESSAGE_FROM_SYSTEM | FORMAT_MESSAGE_IGNORE_INSERTS,
        NULL, code, 0, buffer, sizeof(buffer), NULL);
    while (length > 0 && (buffer[length - 1] == '\r' || buffer[length - 1] == '\n'))
        buffer[--length] = '\0';
    std::ostringstream out;
    out << code << " (0x" << std::hex << code << std::dec << ")";
    if (length > 0)
        out << " " << buffer;
    return out.str();
}

inline std::string ptrText(const void* value)
{
    std::ostringstream out;
    out << value;
    return out.str();
}

class Report
{
public:
    explicit Report(const std::string& path)
        : mFile(path.c_str(), std::ios::out | std::ios::trunc)
    {
    }

    void line(const std::string& text)
    {
        std::cout << text << std::endl;
        if (mFile)
            mFile << text << std::endl;
    }

    void section(const std::string& title)
    {
        line("");
        line("== " + title);
    }

private:
    std::ofstream mFile;
};
```

`class_ogrewindow.cpp`:

```cpp
#include "win32_report.h"

#include <cstring>
#include <string>

namespace
{
std::string exeDirectory()
{
    char path[MAX_PATH];
    DWORD length = GetModuleFileNameA(NULL, path, MAX_PATH);
    std::string full(path, length);
    std::string::size_type slash = full.find_last_of("\\/");
    return slash == std::string::npos ? "." : full.substr(0, slash);
}

void oneCycle(Report& report, const char* title, const char* className, HINSTANCE instance)
{
    WNDPROC proc = DefWindowProcA;
    report.section(title);
    report.line(std::string("class=") + className);
    report.line("style=CS_OWNDC window=WS_VISIBLE|WS_CLIPCHILDREN|WS_OVERLAPPEDWINDOW");
    report.line("wndproc=" + ptrText(reinterpret_cast<const void*>(proc))
        + " wndproc_module=" + modulePath(moduleFromAddress(reinterpret_cast<const void*>(proc))));
    report.line("hinst=" + ptrText(instance) + " hinst_module=" + modulePath(instance));

    WNDCLASSA windowClass;
    std::memset(&windowClass, 0, sizeof(windowClass));
    windowClass.style = CS_OWNDC;
    windowClass.lpfnWndProc = proc;
    windowClass.hInstance = instance;
    windowClass.hIcon = LoadIconA(NULL, IDI_APPLICATION);
    windowClass.hCursor = LoadCursorA(NULL, IDC_ARROW);
    windowClass.hbrBackground = static_cast<HBRUSH>(GetStockObject(BLACK_BRUSH));
    windowClass.lpszClassName = className;

    SetLastError(0);
    ATOM atom = RegisterClassA(&windowClass);
    report.line(atom ? "RegisterClass ok" : "RegisterClass FAIL error=" + win32Message(GetLastError()));

    SetLastError(0);
    HWND hwnd = CreateWindowExA(0, className, "OgreWin32Window module-handle test",
        WS_VISIBLE | WS_CLIPCHILDREN | WS_OVERLAPPEDWINDOW,
        CW_USEDEFAULT, CW_USEDEFAULT, 320, 240, NULL, NULL, instance, NULL);
    report.line(hwnd ? "CreateWindowEx ok" : "CreateWindowEx FAIL error=" + win32Message(GetLastError()));

    if (hwnd)
    {
        SetLastError(0);
        report.line(DestroyWindow(hwnd) ? "DestroyWindow ok" : "DestroyWindow FAIL error=" + win32Message(GetLastError()));
    }

    SetLastError(0);
    report.line(UnregisterClassA(className, instance)
        ? "UnregisterClass ok"
        : "UnregisterClass FAIL error=" + win32Message(GetLastError()));
}
}

int main()
{
    std::string probePath = exeDirectory() + "\\probe.dll";
    Report report(exeDirectory() + "\\class-ogrewindow.txt");
    HMODULE probe = LoadLibraryA(probePath.c_str());
    if (!probe)
    {
        report.line("LoadLibrary probe FAIL error=" + win32Message(GetLastError()));
        return 2;
    }
    HMODULE user32 = GetModuleHandleA("user32.dll");
    report.line("user32=" + modulePath(user32));
    report.line("probe=" + modulePath(probe));
    report.line("DefWindowProc is the default OgreWin32Window procedure and lives in user32.dll");

    oneCycle(report, "A. NULL hinst", "OgreAbGLWindowNull", NULL);
    oneCycle(report, "B. hinst is the module containing DefWindowProc", "OgreAbGLWindowUser", user32);
    oneCycle(report, "C. hinst is another loaded DLL", "OgreAbGLWindowProbe", probe);
    report.section("done");
    return 0;
}
```
